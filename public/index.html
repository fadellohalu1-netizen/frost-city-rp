require("dotenv").config(); const express = require("express"); const session = require("express-session"); const passport = require("passport"); const GoogleStrategy = require("passport-google-oauth20").Strategy; const bcrypt = require("bcryptjs"); const helmet = require("helmet"); const { rateLimit } = require("express-rate-limit"); const fs = require("fs"); const path = require("path"); const crypto = require("crypto"); const app = express(); const PORT = process.env.PORT || 3000; const isProduction = process.env.NODE_ENV === "production"; app.set("trust proxy", 1); app.use(helmet({ contentSecurityPolicy: false })); app.use(express.json({ limit: "2mb" })); // จำกัดจำนวนครั้งในการลองเข้าสู่ระบบและส่งรหัส const authLimiter = rateLimit({ windowMs: 15 * 60 * 1000, limit: 20, standardHeaders: "draft-7", legacyHeaders: false }); app.use("/auth", authLimiter); app.use("/api/access/redeem", authLimiter); // ------------------------------ // DATABASE (JSON) // ------------------------------ const dataDir = path.join(__dirname, "data"); const databaseFile = path.join(dataDir, "database.json"); if (!fs.existsSync(dataDir)) { fs.mkdirSync(dataDir, { recursive: true }); } function loadDatabase() { if (!fs.existsSync(databaseFile)) { return { panels: [], applications: [], users: [], accessCodes: [], auditLogs: [] }; } try { const oldData = JSON.parse( fs.readFileSync(databaseFile, "utf8") ); return { panels: Array.isArray(oldData.panels) ? oldData.panels : [], applications: Array.isArray(oldData.applications) ? oldData.applications : [], users: Array.isArray(oldData.users) ? oldData.users : [], accessCodes: Array.isArray(oldData.accessCodes) ? oldData.accessCodes : [], auditLogs: Array.isArray(oldData.auditLogs) ? oldData.auditLogs : [] }; } catch (error) { console.error("อ่านฐานข้อมูลไม่สำเร็จ:", error.message); throw new Error("ไม่สามารถอ่าน database.json ได้ กรุณาตรวจสอบไฟล์ฐานข้อมูล"); } } let db = loadDatabase(); function saveDatabase() { const tempFile = databaseFile + ".tmp"; fs.writeFileSync( tempFile, JSON.stringify(db, null, 2), "utf8" ); fs.renameSync(tempFile, databaseFile); } function audit(action, user, details = {}) { db.auditLogs.push({ id: crypto.randomUUID(), action, user: user?.email || "system", role: user?.role || "system", details, createdAt: new Date().toISOString() }); // เก็บประวัติล่าสุดไม่เกิน 2,000 รายการ if (db.auditLogs.length > 2000) { db.auditLogs = db.auditLogs.slice(-2000); } saveDatabase(); } // ------------------------------ // SESSION // ------------------------------ if (isProduction && !process.env.SESSION_SECRET) { throw new Error("กรุณาตั้งค่า SESSION_SECRET ใน Environment Variables"); } app.use(session({ name: "frost.sid", secret: process.env.SESSION_SECRET || "local-development-secret-change-this", resave: false, saveUninitialized: false, cookie: { httpOnly: true, secure: isProduction, sameSite: "lax", maxAge: 1000 * 60 * 60 * 8 } })); // ------------------------------ // GOOGLE LOGIN // ------------------------------ passport.serializeUser((user, done) => { done(null, user.email); }); passport.deserializeUser((email, done) => { const user = db.users.find( item => item.email === email ); done(null, user || false); }); app.use(passport.initialize()); app.use(passport.session()); const googleConfigured = process.env.GOOGLE_CLIENT_ID && process.env.GOOGLE_CLIENT_SECRET; if (googleConfigured) { passport.use(new GoogleStrategy({ clientID: process.env.GOOGLE_CLIENT_ID, clientSecret: process.env.GOOGLE_CLIENT_SECRET, callbackURL: process.env.GOOGLE_CALLBACK_URL || "http://localhost:3000/auth/google/callback" }, (accessToken, refreshToken, profile, done) => { const email = profile.emails?.[0]?.value?.toLowerCase(); if (!email || profile.emails?.[0]?.verified !== true) { return done(null, false); } let user = db.users.find( item => item.email === email ); // บัญชีนายกกำหนดจาก Environment Variable เท่านั้น if (email === (process.env.MAYOR_EMAIL || "").toLowerCase()) { if (!user) { user = { email, name: profile.displayName || email, role: "mayor", applicationAccess: true, staffId: null, createdAt: new Date().toISOString() }; db.users.push(user); } else { user.role = "mayor"; user.applicationAccess = true; } } else if (!user) { user = { email, name: profile.displayName || email, role: "guest", applicationAccess: false, staffId: null, createdAt: new Date().toISOString() }; db.users.push(user); } else { user.name = profile.displayName || user.name; } saveDatabase(); return done(null, user); })); } // ------------------------------ // AUTHENTICATION HELPERS // ------------------------------ function requireLogin(req, res, next) { if (!req.isAuthenticated || !req.isAuthenticated()) { return res.status(401).json({ success: false, message: "กรุณาเข้าสู่ระบบก่อน" }); } next(); } function requireRole(...roles) { return (req, res, next) => { if (!req.isAuthenticated || !req.isAuthenticated()) { return res.status(401).json({ success: false, message: "กรุณาเข้าสู่ระบบก่อน" }); } if (!roles.includes(req.user.role)) { return res.status(403).json({ success: false, message: "คุณไม่มีสิทธิ์ใช้งานส่วนนี้" }); } next(); }; } function requireApplicationAccess(req, res, next) { if (!req.isAuthenticated || !req.isAuthenticated()) { return res.status(401).json({ success: false, message: "กรุณาเข้าสู่ระบบ Google ก่อน" }); } if ( req.user.role !== "mayor" && !req.user.applicationAccess ) { return res.status(403).json({ success: false, message: "บัญชีนี้ยังไม่ได้รับอนุญาตให้สมัครทีมงาน" }); } next(); } // ------------------------------ // GOOGLE AUTH ROUTES // ------------------------------ app.get("/auth/google", (req, res, next) => { if (!googleConfigured) { return res.status(503).send( "ระบบ Google Login ยังไม่ได้ตั้งค่า" ); } passport.authenticate("google", { scope: ["profile", "email"] })(req, res, next); }); app.get( "/auth/google/callback", (req, res, next) => { passport.authenticate("google", { failureRedirect: "/?login=failed" })(req, res, next); }, (req, res) => { req.session.save(() => { res.redirect("/"); }); } ); app.post("/auth/logout", (req, res, next) => { req.logout(error => { if (error) return next(error); req.session.destroy(() => { res.clearCookie("frost.sid"); res.json({ success: true }); }); }); }); app.get("/api/me", (req, res) => { if (!req.isAuthenticated || !req.isAuthenticated()) { return res.json({ loggedIn: false, user: null }); } res.json({ loggedIn: true, user: { email: req.user.email, name: req.user.name, role: req.user.role, applicationAccess: req.user.applicationAccess, staffId: req.user.staffId } }); }); // ------------------------------ // ACCESS CODE // ------------------------------ // นายกออก Access Code ให้บัญชี Google ที่ระบุ app.post( "/api/access-codes", requireRole("mayor"), async (req, res) => { try { const email = String(req.body.email || "") .trim() .toLowerCase(); if (!/^[^\s@]+@[^\s@]+\.[^\s@]+$/.test(email)) { return res.status(400).json({ success: false, message: "กรุณาระบุอีเมลให้ถูกต้อง" }); } const code = crypto.randomBytes(5) .toString("hex") .toUpperCase(); const codeHash = await bcrypt.hash(code, 10); db.accessCodes.push({ id: crypto.randomUUID(), email, codeHash, used: false, createdBy: req.user.email, createdAt: new Date().toISOString() }); saveDatabase(); audit("ACCESS_CODE_CREATED", req.user, { email }); // แสดงรหัสให้เห็นเพียงครั้งเดียว res.json({ success: true, email, accessCode: code, message: "ส่งรหัสนี้ให้เจ้าของบัญชี Google ที่ระบุ" }); } catch (error) { console.error(error); res.status(500).json({ success: false, message: "สร้าง Access Code ไม่สำเร็จ" }); } } ); // ผู้สมัครนำรหัสมาแลกสิทธิ์ app.post("/api/access/redeem", requireLogin, async (req, res) => { try { const code = String(req.body.code || "") .trim() .toUpperCase(); if (!code) { return res.status(400).json({ success: false, message: "กรุณากรอก Access Code" }); } const availableCodes = db.accessCodes.filter(item => item.email === req.user.email && !item.used ); let matchedCode = null; for (const item of availableCodes) { if (await bcrypt.compare(code, item.codeHash)) { matchedCode = item; break; } } if (!matchedCode) { return res.status(400).json({ success: false, message: "รหัสไม่ถูกต้อง หรือไม่ตรงกับบัญชี Google นี้" }); } matchedCode.used = true; matchedCode.usedAt = new Date().toISOString(); const user = db.users.find( item => item.email === req.user.email ); user.applicationAccess = true; saveDatabase(); audit("ACCESS_CODE_REDEEMED", user); res.json({ success: true, message: "ยืนยันสิทธิ์สำเร็จ สามารถเข้าสู่ระบบสมัครทีมงานได้" }); } catch (error) { console.error(error); res.status(500).json({ success: false, message: "ตรวจสอบรหัสไม่สำเร็จ" }); } }); // ------------------------------ // PUBLIC STATUS // ------------------------------ app.get("/api/status", (req, res) => { res.json({ server: "FROST CITY RP", status: "online", system: "Secure Staff Center", googleLogin: Boolean(googleConfigured) }); }); // ------------------------------ // EXAM PANELS // ------------------------------ // ผู้สมัครเห็นโจทย์และตัวเลือก แต่ไม่เห็นเฉลย app.get("/api/panels", (req, res) => { const safePanels = db.panels.map(panel => ({ ...panel, questions: (panel.questions || []).map(question => { const { correct, answer, ...safeQuestion } = question; return safeQuestion; }) })); res.json(safePanels); }); // เฉพาะนายกและแอดมินสร้างข้อสอบได้ app.post( "/api/panels", requireRole("mayor", "admin"), (req, res) => { const { name, description, questions, passMark } = req.body; if ( typeof name !== "string" || !name.trim() || !Array.isArray(questions) || questions.length === 0 || questions.length > 100 ) { return res.status(400).json({ success: false, message: "ข้อมูลข้อสอบไม่ถูกต้อง" }); } for (const question of questions) { if ( typeof question.text !== "string" || !Array.isArray(question.options) || question.options.length !== 4 || !Number.isInteger(question.correct) || question.correct < 0 || question.correct > 3 ) { return res.status(400).json({ success: false, message: "รูปแบบคำถามหรือเฉลยไม่ถูกต้อง" }); } } const panel = { id: crypto.randomUUID(), name: name.trim(), description: String(description || ""), questions: questions.map(question => ({ text: question.text, options: question.options.map(String), correct: question.correct, points: Math.max(1, Math.min(100, Number(question.points) || 5)) })), passMark: Math.max(0, Math.min(100, Number(passMark) || 70)), createdAt: new Date().toISOString(), createdBy: req.user.email }; db.panels.push(panel); saveDatabase(); audit("PANEL_CREATED", req.user, { panelId: panel.id }); res.json({ success: true, panel: { id: panel.id, name: panel.name } }); } ); // ------------------------------ // APPLICATIONS // ------------------------------ // สมัครได้เฉพาะบัญชีที่ได้รับ Access Code แล้ว app.post( "/api/applications", requireApplicationAccess, (req, res) => { const { panelId, nickname, icName, answers } = req.body; const panel = db.panels.find( item => item.id === panelId ); if (!panel) { return res.status(404).json({ success: false, message: "ไม่พบชุดข้อสอบนี้" }); } if ( typeof nickname !== "string" || !nickname.trim() || typeof icName !== "string" || !icName.trim() || !Array.isArray(answers) || answers.length !== panel.questions.length ) { return res.status(400).json({ success: false, message: "กรุณากรอกข้อมูลและตอบข้อสอบให้ครบ" }); } let score = 0; let total = 0; for (let i = 0; i < panel.questions.length; i++) { const question = panel.questions[i]; const selected = answers[i]; const points = Number(question.points) || 5; total += points; if ( !Number.isInteger(selected) || selected < 0 || selected >= question.options.length ) { return res.status(400).json({ success: false, message: `คำตอบข้อ ${i + 1} ไม่ถูกต้อง` }); } if (selected === question.correct) { score += points; } } const percent = total > 0 ? Math.round((score / total) * 100) : 0; const passed = percent >= panel.passMark; const application = { id: crypto.randomUUID(), email: req.user.email, panelId: panel.id, nickname: nickname.trim(), icName: icName.trim(), answers, score, total, percent, passed, status: passed ? "passed" : "failed", createdAt: new Date().toISOString() }; db.applications.push(application); saveDatabase(); audit("APPLICATION_SUBMITTED", req.user, { applicationId: application.id, passed }); res.json({ success: true, result: { score, total, percent, passed, message: passed ? "ผ่านการทดสอบ" : "ไม่ผ่านการทดสอบ" } }); } ); // ข้อมูลใบสมัครเป็นข้อมูลภายในเท่านั้น app.get( "/api/applications", requireRole("mayor", "admin"), (req, res) => { const applications = db.applications.map(item => ({ ...item })); res.json(applications); } ); // ------------------------------ // STAFF MANAGEMENT // ------------------------------ // นายกเท่านั้นที่แต่งตั้งตำแหน่งและสร้าง Staff ID app.post( "/api/staff", requireRole("mayor"), (req, res) => { const email = String(req.body.email || "") .trim() .toLowerCase(); const role = String(req.body.role || ""); if (!["admin", "staff"].includes(role)) { return res.status(400).json({ success: false, message: "ตำแหน่งต้องเป็น admin หรือ staff" }); } const user = db.users.find( item => item.email === email ); if (!user) { return res.status(404).json({ success: false, message: "ไม่พบบัญชีนี้ กรุณาให้เจ้าของบัญชีเข้าสู่ระบบ Google ก่อน" }); } if (user.role === "mayor") { return res.status(400).json({ success: false, message: "ไม่สามารถเปลี่ยนตำแหน่งนายกเมืองได้" }); } const prefix = role === "admin" ? "ADFC" : "STFC"; const existingIds = db.users .map(item => item.staffId) .filter(Boolean); let number = 1; let staffId; do { staffId = `${prefix}-${String(number).padStart(4, "0")}`; number++; } while (existingIds.includes(staffId)); user.role = role; user.staffId = staffId; user.applicationAccess = true; user.appointedAt = new Date().toISOString(); saveDatabase(); audit("STAFF_APPOINTED", req.user, { email, role, staffId }); res.json({ success: true, staff: { email: user.email, name: user.name, role: user.role, staffId: user.staffId } }); } ); // ------------------------------ // PUBLIC STAFF VERIFICATION // ------------------------------ // เปิดให้ทุกคนตรวจสอบได้ แต่แสดงเฉพาะข้อมูลสาธารณะ app.get("/api/verify/:staffId", (req, res) => { const staff = db.users.find( item => item.staffId && item.staffId.toLowerCase() === req.params.staffId.toLowerCase() && ["admin", "staff"].includes(item.role) ); if (!staff) { return res.status(404).json({ verified: false, message: "ไม่พบข้อมูลทีมงาน" }); } res.json({ verified: true, staff: { name: staff.name, staffId: staff.staffId, role: staff.role } }); }); // ------------------------------ // DASHBOARD / AUDIT // ------------------------------ app.get( "/api/audit-logs", requireRole("mayor"), (req, res) => { res.json(db.auditLogs.slice(-200)); } ); app.get( "/api/dashboard", requireRole("mayor", "admin"), (req, res) => { const applications = db.applications; res.json({ panels: db.panels.length, applications: applications.length, passed: applications.filter(item => item.passed).length, failed: applications.filter(item => !item.passed).length }); } ); // ------------------------------ // FRONTEND // ------------------------------ app.use(express.static(path.join(__dirname, "public"))); app.get("/", (req, res) => { res.sendFile( path.join(__dirname, "public", "index.html") ); }); // ------------------------------ // ERROR HANDLER // ------------------------------ app.use((error, req, res, next) => { console.error("Server error:", error); res.status(500).json({ success: false, message: "เกิดข้อผิดพลาดภายในเซิร์ฟเวอร์" }); }); // ------------------------------ // START SERVER // ------------------------------ app.listen(PORT, () => { console.log(`FROST CITY RP running on port ${PORT}`); console.log(`Google Login: ${googleConfigured ? "Configured" : "Not configured"}`); });
